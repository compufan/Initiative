import { describe, expect, it } from 'vitest';

import { flutmaske } from '../stickers/engines/flutung.js';
import type { InhaltsTeil } from './bildweise.js';
import type { GelesenesBild } from './bilderLesen.js';
import { folgeTeile } from './folgeTeile.js';

/*
 * Messung (nicht für das Repo): Zwei Quadrate, A wandert nach rechts, B nach unten.
 * Zeichenreihenfolge: A, dann B (B liegt oben).
 */
const B = 320;
const H = 180;
const KANTE = 40;
type Farbe = [number, number, number];

interface Film {
  readonly a: (n: number) => { x: number; y: number } | null;
  readonly b: (n: number) => { x: number; y: number } | null;
  readonly farbeA: Farbe;
  readonly farbeB: Farbe;
  readonly obenA?: boolean;
}

function bild(n: number, film: Film): GelesenesBild {
  const data = new Uint8ClampedArray(B * H * 4);
  const a = film.a(n);
  const b = film.b(n);
  for (let y = 0; y < H; y += 1) {
    for (let x = 0; x < B; x += 1) {
      const at = (y * B + x) * 4;
      const g = 128 + 40 * Math.sin(x / 5.3 + Math.cos(y / 7.1)) * Math.cos(y / 4.7 - x / 13);
      void g;
      let c: Farbe = [36, 64, 92];
      const inA = !!a && x >= a.x && x < a.x + KANTE && y >= a.y && y < a.y + KANTE;
      const inB = !!b && x >= b.x && x < b.x + KANTE && y >= b.y && y < b.y + KANTE;
      if (film.obenA) {
        if (inB) c = film.farbeB;
        if (inA) c = film.farbeA;
      } else {
        if (inA) c = film.farbeA;
        if (inB) c = film.farbeB;
      }
      data[at] = c[0];
      data[at + 1] = c[1];
      data[at + 2] = c[2];
      data[at + 3] = 255;
    }
  }
  return { zeitMs: n * 40, daten: { data, width: B, height: H, colorSpace: 'srgb' } as ImageData };
}

function wahrheit(n: number, film: Film, wer: 'a' | 'b' | 'ab'): Uint8Array {
  const m = new Uint8Array(B * H);
  const a = film.a(n);
  const b = film.b(n);
  for (let y = 0; y < H; y += 1) {
    for (let x = 0; x < B; x += 1) {
      const roh = (p: { x: number; y: number } | null) => !!p && x >= p.x && x < p.x + KANTE && y >= p.y && y < p.y + KANTE;
      const inB = roh(b) && !(film.obenA && roh(a));
      const inA = roh(a) && !(!film.obenA && roh(b));
      if ((wer !== 'b' && inA) || (wer !== 'a' && inB)) m[y * B + x] = 255;
    }
  }
  return m;
}

function deckung(a: Uint8Array, b: Uint8Array): number {
  let s = 0;
  let v = 0;
  for (let i = 0; i < a.length; i += 1) {
    const x = a[i] >= 128;
    const y = b[i] >= 128;
    if (x && y) s += 1;
    if (x || y) v += 1;
  }
  return v === 0 ? 1 : s / v;
}

function tippTeil(id: string, punkte: { x: number; y: number }[]): InhaltsTeil {
  return {
    bereich: 'b1',
    art: 'tipp',
    teil: {
      id,
      modus: 'dazu',
      umkehren: false,
      art: 'tipp',
      mitNetz: false,
      punkte,
      toleranz: 25,
      breite: B,
      hoehe: H,
      alpha: new Uint8Array(B * H),
      marke: 1,
    },
  };
}

const kreuzend: Film = {
  a: (n) => ({ x: 20 + 4 * n, y: 50 }),
  b: (n) => ({ x: 140, y: 10 + 2 * n }),
  farbeA: [16, 112, 48],
  farbeB: [240, 176, 64],
};
const getrennt: Film = { ...kreuzend, b: (n) => ({ x: 230, y: 10 + 2 * n }) };

async function lauf(film: Film, teile: InhaltsTeil[], anzahl = 50) {
  const bilder = Array.from({ length: anzahl }, (_, n) => bild(n, film));
  const { jeBild } = await folgeTeile(bilder, { teile, schluesselAbstand: 4 });
  return { bilder, jeBild };
}

function zeile(werte: number[]) {
  return werte.map((w) => w.toFixed(2)).join(' ');
}


function schwerpunkt(m: Uint8Array): string {
  let sx = 0; let sy = 0; let n = 0;
  for (let y = 0; y < H; y += 1) for (let x = 0; x < B; x += 1) if (m[y * B + x] >= 128) { sx += x; sy += y; n += 1; }
  return n === 0 ? '-' : `${Math.round(sx / n)},${Math.round(sy / n)}`;
}

describe('Diagnose', () => {
  for (const [name, film] of [
    ['B oben (Verdecker B)', kreuzend],
    ['A oben (A verdeckt B, A wird verfolgt)', { ...kreuzend, obenA: true }],
  ] as [string, Film][]) {
    it(`A allein: ${name}`, async () => {
      const a0 = film.a(0)!;
      const teil = tippTeil('tA', [{ x: a0.x + 20, y: a0.y + 20 }]);
      const { jeBild } = await lauf(film, [teil]);
      const werte = jeBild.map((m, n) => deckung(m.get('tA')?.werte ?? new Uint8Array(B * H), wahrheit(n, film, 'a')));
      console.log(`DIAG ${name} Deckung:`, zeile(werte));
      console.log(`DIAG ${name} Schwerpunkt Maske:`, jeBild.map((m) => schwerpunkt(m.get('tA')?.werte ?? new Uint8Array(B * H))).slice(18, 50).join(' '));
      console.log(`DIAG ${name} Schwerpunkt wahr A :`, jeBild.map((_, n) => schwerpunkt(wahrheit(n, film, 'a'))).slice(18, 50).join(' '));
    }, 120_000);
  }
  it('B wird verdeckt (A oben), B wird verfolgt', async () => {
    const film = { ...kreuzend, obenA: true };
    const b0 = film.b(0)!;
    const teil = tippTeil('tB', [{ x: b0.x + 20, y: b0.y + 20 }]);
    const { jeBild } = await lauf(film, [teil]);
    const werte = jeBild.map((m, n) => deckung(m.get('tB')?.werte ?? new Uint8Array(B * H), wahrheit(n, film, 'b')));
    console.log('DIAG B verdeckt Deckung:', zeile(werte));
  }, 120_000);
});
